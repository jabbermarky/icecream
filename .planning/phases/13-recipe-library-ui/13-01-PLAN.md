   - Wire up click handler in recipe button initialization:
     ```javascript
     ui.recipe.btnRecipeLibrary.onclick = () => {
         showRecipeLibrary(recipeStorage, {
             onLoad: (name) => {
                 // Load recipe from storage and set it
                 recipeStorage.loadRecipe(name).then(data => {
                     if (data) {
                         setRecipe(data.data.Recipe, data.data.Ingredients);
                         Info(`Loaded "${name}" from library`);
                     }
                 });
             },
             onDelete: (name) => {
                 // Delete callback will be fully wired in 13-02
                 console.log('Delete requested for:', name);
             }
         });
     };
     ```
