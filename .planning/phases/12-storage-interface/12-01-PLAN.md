```javascript
  // Save recipe - returns Promise<void>
```

Export a `createStorage(implementation)` factory function that validates the implementation has all required methods.

**js/storage/indexeddb-storage.js** - IndexedDB implementation using idb:
- Import `openDB` from 'https://esm.sh/idb@8' (ESM CDN, no npm install needed)
- Database name: 'ice-ed-recipes', version: 1
- Object store: 'recipes' with keyPath: 'name'
- Index on 'updatedAt' for sorting by recent
- Each recipe stored as: `{ name, updatedAt, data: { Recipe, Ingredients } }`
- Export `IndexedDBStorage` object implementing StorageInterface
- Include `initIndexedDBStorage()` that opens DB and returns storage instance

Handle errors gracefully - wrap IndexedDB operations in try/catch, return null or empty array on failure rather than throwing (consistent with parseRecipeFile pattern in file-io.js).

```javascript
async testRecipeStorage() {
  // Test that storage is initialized
  const storage = await this.page.evaluate(() => {
    return typeof window.getRecipeStorage === 'function' && window.getRecipeStorage() !== null;
  });

  if (!storage) {
    throw new Error('Recipe storage not initialized');
  }

  // Test save/load/list/delete cycle
  const testRecipe = {
    name: 'Test Storage Recipe',
    data: {
      Recipe: {
        Name: 'Test Storage Recipe',
        Notes: 'Test notes',
        Type: 'Standard',
        ServingTemperature: -18,
        Hardness: 0.75,
        Overrun: 0.3,
        Ingredients: []
      },
      Ingredients: {}
    }
  };

  // Save
  await this.page.evaluate(async (recipe) => {
    const storage = window.getRecipeStorage();
    await storage.saveRecipe(recipe);
  }, testRecipe);

  // List - should include test recipe
  const listAfterSave = await this.page.evaluate(async () => {
    const storage = window.getRecipeStorage();
    return await storage.listRecipes();
  });

  if (!listAfterSave.some(r => r.name === 'Test Storage Recipe')) {
    throw new Error('Recipe not found in list after save');
  }

  // Load
  const loaded = await this.page.evaluate(async () => {
    const storage = window.getRecipeStorage();
    return await storage.loadRecipe('Test Storage Recipe');
  });

  if (!loaded || loaded.data.Recipe.Name !== 'Test Storage Recipe') {
    throw new Error('Failed to load saved recipe');
  }

  // Delete
  await this.page.evaluate(async () => {
    const storage = window.getRecipeStorage();
    await storage.deleteRecipe('Test Storage Recipe');
  });

  // Verify deleted
  const listAfterDelete = await this.page.evaluate(async () => {
    const storage = window.getRecipeStorage();
    return await storage.listRecipes();
  });

  if (listAfterDelete.some(r => r.name === 'Test Storage Recipe')) {
    throw new Error('Recipe still exists after delete');
  }

  console.log('Recipe storage test passed');
}
```
